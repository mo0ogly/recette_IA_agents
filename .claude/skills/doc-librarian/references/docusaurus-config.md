## docusaurus.config.js

```js
// @ts-check
const { themes } = require('prism-react-renderer');

/** @type {import('@docusaurus/types').Config} */
const config = {
  title: 'LIA-SEC Documentation',
  tagline: 'Security Audit Platform',
  favicon: 'img/favicon.ico',

  url: 'https://inspect-ia-agents.tech',
  baseUrl: '/docs/',               // CRITICAL: served at /docs/ not /

  organizationName: 'lia-sec',
  projectName: 'lia-sec-docs',

  onBrokenLinks: 'warn',           // warn not throw (some GAP docs may not exist yet)
  onBrokenMarkdownLinks: 'warn',

  i18n: {
    defaultLocale: 'en',
    locales: ['en'],
  },

  markdown: {
    mermaid: true,                 // Enable Mermaid in markdown (Category B runtime)
  },

  themes: ['@docusaurus/theme-mermaid'],

  plugins: [
    // Process SVG files (Category A pre-rendered diagrams)
    '@svgr/webpack',
  ],

  themeConfig:
    /** @type {import('@docusaurus/preset-classic').ThemeConfig} */
    ({
      navbar: {
        title: 'LIA-SEC Docs',
        logo: {
          alt: 'LIA-SEC Logo',
          src: 'img/logo.svg',
        },
        items: [
          { type: 'docSidebar', sidebarId: 'apiSidebar', label: 'API', position: 'left' },
          { type: 'docSidebar', sidebarId: 'archSidebar', label: 'Architecture', position: 'left' },
          { type: 'docSidebar', sidebarId: 'techSidebar', label: 'Technical', position: 'left' },
          { type: 'docSidebar', sidebarId: 'userSidebar', label: 'User Guides', position: 'left' },
          { type: 'docSidebar', sidebarId: 'secSidebar', label: 'Security', position: 'left' },
          { type: 'docSidebar', sidebarId: 'specSidebar', label: 'Specs', position: 'left' },
        ],
      },

      // Dark theme matching LIA-SEC CyberTheme
      colorMode: {
        defaultMode: 'dark',
        disableSwitch: true,       // LIA-SEC is dark-only (audit/RSSI audience)
        respectPrefersColorScheme: false,
      },

      prism: {
        theme: themes.dracula,
        darkTheme: themes.dracula,
        additionalLanguages: ['go', 'bash', 'sql', 'yaml', 'typescript'],
      },

      mermaid: {
        theme: { light: 'dark', dark: 'dark' },   // Always dark
        options: {
          fontFamily: 'JetBrains Mono, monospace',
          fontSize: 14,
        },
      },

      footer: {
        style: 'dark',
        copyright: `LIA-SEC Platform Documentation`,
      },
    }),

  presets: [
    [
      'classic',
      /** @type {import('@docusaurus/preset-classic').Options} */
      ({
        docs: {
          routeBasePath: '/',           // Docs at /docs/ (baseUrl already set to /docs/)
          sidebarPath: require.resolve('./sidebars.js'),
          editUrl: undefined,           // No edit links (internal repo)
        },
        blog: false,                    // No blog
        theme: {
          customCss: require.resolve('./src/css/custom.css'),
        },
      }),
    ],
  ],
};

module.exports = config;
```

## Scaffold Commands (Vague 1)

```bash
# Create docs-site/ from scratch
cd /opt/lia-sec/dev
npx create-docusaurus@3.7.0 docs-site classic --typescript false

# Install mermaid theme
cd docs-site
npm install @docusaurus/theme-mermaid@3.7.0

# Replace generated config files with LIA-SEC versions (above)
# Then link docs directory
ln -s ../_alire/01_DOCUMENTATION docs/

# Test locally (port 3001 -- free port)
npm run start -- --port 3001

# Build
npm run build

# Pagefind index
npx pagefind --source build
```

## Handling .doc-meta.json Files

Docusaurus will attempt to process ALL files in the docs/ directory.
The `.doc-meta.json` sidecar files MUST be excluded from processing:

```js
// In docusaurus.config.js, add to docs preset:
docs: {
  exclude: ['**/*.doc-meta.json', '**/archive/**', '**/exports/**'],
  // ...
}
```
